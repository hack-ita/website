"width" 1200 "height" 800) }} Performance note: images that are ALREADY .webp and need no resizing are served as-is (published directly, no re-encode). Previously every image was decoded and re-encoded to WebP twice on every build — the main cause of slow Netlify builds, since ~540 of ~560 article images are already WebP. Only non-webp sources, or sources that need resizing, are processed now. */}} {{ $params := . }} {{ $src := $params.src }} {{ $alt := $params.alt | default "" }} {{ $class := $params.class | default "" }} {{ $width := $params.width | default 0 }} {{ $height := $params.height | default 0 }} {{ $highFetchPrio := $params.highFetchPrio | default false }} {{ $ext := lower (strings.TrimPrefix (path.Ext $src) ".") }} {{ $img := resources.Get $src }} {{ if $img }} {{ $needsResize := or (gt $width 0) (gt $height 0) }} {{ if $needsResize }} {{/* Resize + ensure WebP output in a single encode */}} {{ $img = ($img.Fit (printf "%dx%d" $width $height)).Process "webp" }} {{ else if ne $ext "webp" }} {{/* Convert non-webp sources to WebP once */}} {{ $img = $img.Process "webp" }} {{/* else: already WebP + no resize -> serve original, no re-encode */}} {{ end }} {{ else }} {{ warnf "Image not found: %s" $src }} {{ end }}
